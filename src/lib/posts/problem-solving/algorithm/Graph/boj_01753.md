# [백준 : 1753. 최단경로](https://www.acmicpc.net/problem/1753)

## **목표**

> 시작점에서 모든 노드까지의 최단 거리를 구하는 문제.
> `방향 그래프의 node A에서 다른 노드까지의 거리를 구하는 문제. 입력 크기를 보면 node = 20000에 edges = 300000이라 다익스트라로 풀이`

## **문제 풀이**

### **Python**

[tabs: Solution 1, Solution 2]

```python
from collections import defaultdict as dd
import heapq as hq
import sys

input = lambda : sys.stdin.readline().rstrip()

N,K = map(int,input().split())
start = int(input())

# 방향 그래프 : edges[a][b] = cost (a->b로 가는 cost 최소값으로 갱신하기)
edges = dd(lambda : dd(lambda : float('inf')))
for _ in range(K):
    a,b,cost = map(int,input().split())
    edges[a][b] = min(edges[a][b],cost)

V = [float('inf')]*(N+1)
V[start] = 0

heap = [(0,start)]
while heap:

    c,x = hq.heappop(heap)
    for nx,cost in edges[x].items():
        nc = c+cost
        if nc < V[nx]:
            V[nx] = nc
            hq.heappush(heap,(nc,nx))

for cost in V[1:]:
    print(cost if cost != float('inf') else 'INF')
```

```python
from dataclasses import dataclass
from collections import defaultdict as dd
import heapq as hq
import sys

input = lambda : sys.stdin.readline().rstrip()
INF = float('inf')

N,M = map(int,input().split())
start = int(input())

G = dd(lambda : dd(lambda : float('inf')))
for _ in range(M):
    a,b,cost = map(int,input().split())
    G[a][b] = min(G[a][b],cost)

V = [INF]*(N+1)

@dataclass(frozen=True,order=True)
class Vertex:
    cum_cost: int
    x: int

    @property
    def next_step(self):
        return [Vertex(self.cum_cost + cost, nx) for nx,cost in G[self.x].items()]
    
    def visit(self):
        V[self.x] = self.cum_cost
        
    def is_valid(self):
        return self.cum_cost < V[self.x]

def di(sv: Vertex):
    V[sv.x] = 0
    heap = [sv]
    
    while heap:
        cur_v = hq.heappop(heap)
        
        if cur_v.cum_cost > V[cur_v.x]:
            continue
            
        for nxt_v in cur_v.next_step:
            if nxt_v.is_valid():
                nxt_v.visit()
                hq.heappush(heap,nxt_v)

di(Vertex(0,start))
for cost in V[1:]:
    print(cost if cost != INF else 'INF')
```
